// ===== STORAGE =====
let products = JSON.parse(localStorage.getItem('fgeneral_products')) || [];
let currentUser = JSON.parse(sessionStorage.getItem('fgeneral_user')) || null;

// ===== DOM REFS =====
const loginForm = document.getElementById('loginForm');
const signupForm = document.getElementById('signupForm');

// ===== LOGIN =====
loginForm?.addEventListener('submit', function(e) {
    e.preventDefault();
    const email = document.getElementById('loginEmail').value.trim();
    const password = document.getElementById('loginPassword').value;

    if (!email || !password) {
        showAIMessage('Please fill in all fields.', 'error');
        return;
    }

    // Check if user exists
    const users = JSON.parse(localStorage.getItem('fgeneral_users')) || [];
    const user = users.find(u => 
        (u.email === email || u.phone === email) && u.password === password
    );

    if (user) {
        sessionStorage.setItem('fgeneral_user', JSON.stringify(user));
        window.location.href = 'dashboard.html';
    } else {
        showAIMessage('❌ Invalid credentials. Please check your email/phone and password.');
    }
});

// ===== SIGNUP =====
signupForm?.addEventListener('submit', function(e) {
    e.preventDefault();
    const name = document.getElementById('signupName').value.trim();
    const email = document.getElementById('signupEmail').value.trim();
    const password = document.getElementById('signupPassword').value;

    if (!name || !email || password.length < 6) {
        showAIMessage('⚠️ Name, email/phone, and password (min 6 chars) required.');
        return;
    }

    const users = JSON.parse(localStorage.getItem('fgeneral_users')) || [];
    if (users.find(u => u.email === email || u.phone === email)) {
        showAIMessage('⚠️ Account already exists. Please login.');
        return;
    }

    const newUser = { name, email, password, phone: email };
    users.push(newUser);
    localStorage.setItem('fgeneral_users', JSON.stringify(users));
    sessionStorage.setItem('fgeneral_user', JSON.stringify(newUser));
    showAIMessage('✅ Account created! Welcome to F. General Tech!');
    setTimeout(() => window.location.href = 'dashboard.html', 1500);
});

// ===== TOGGLE LOGIN/SIGNUP =====
function showSignup() {
    loginForm.style.display = 'none';
    signupForm.style.display = 'block';
}
function showLogin() {
    loginForm.style.display = 'block';
    signupForm.style.display = 'none';
}

// ===== TOGGLE PASSWORD =====
function togglePassword(id) {
    const input = document.getElementById(id);
    input.type = input.type === 'password' ? 'text' : 'password';
}

// ===== LOGOUT =====
function logout() {
    sessionStorage.removeItem('fgeneral_user');
    window.location.href = 'index.html';
}

// ===== DASHBOARD - LOAD PRODUCTS =====
function loadProducts() {
    const grid = document.getElementById('productGrid');
    if (!grid) return;

    if (products.length === 0) {
        grid.innerHTML = `
            <div style="grid-column:1/-1; text-align:center; padding:60px 20px; color:var(--text-muted);">
                <i class="fas fa-box" style="font-size:3rem; margin-bottom:15px; color:var(--gold);"></i>
                <p style="font-size:1.2rem;">No products yet.</p>
                <p>Click "Add New" to list your first device!</p>
            </div>
        `;
        return;
    }

    grid.innerHTML = products.map((p, i) => `
        <div class="product-card">
            <div class="product-name">${p.name}</div>
            <span class="product-category">${p.category || 'Other'}</span>
            <div class="product-price">₦${p.price || '0'}</div>
            <p style="color:var(--text-muted); font-size:0.85rem; margin:6px 0;">${p.desc || 'No description'}</p>
            <span class="product-status status-${p.status || 'available'}">${(p.status || 'Available').toUpperCase()}</span>
            <div class="product-actions">
                <button class="btn-status" onclick="toggleStatus(${i})">
                    <i class="fas fa-sync"></i> Toggle Status
                </button>
                <button class="btn-delete" onclick="deleteProduct(${i})">
                    <i class="fas fa-trash"></i> Delete
                </button>
            </div>
        </div>
    `).join('');
}

// ===== ADD PRODUCT =====
document.getElementById('addProductForm')?.addEventListener('submit', function(e) {
    e.preventDefault();
    const name = document.getElementById('prodName').value.trim();
    const category = document.getElementById('prodCategory').value;
    const price = document.getElementById('prodPrice').value;
    const desc = document.getElementById('prodDesc').value.trim();
    const status = document.getElementById('prodStatus').value;

    if (!name) {
        showAIMessage('⚠️ Product name is required!');
        return;
    }

    products.push({ 
        id: Date.now(), 
        name, 
        category, 
        price: price || '0', 
        desc: desc || 'No description', 
        status 
    });
    localStorage.setItem('fgeneral_products', JSON.stringify(products));
    this.reset();
    showAIMessage('✅ Product added successfully!');
    loadProducts();
    showSection('products');
});

// ===== TOGGLE STATUS =====
function toggleStatus(index) {
    const p = products[index];
    const statuses = ['available', 'sold', 'unavailable'];
    let current = statuses.indexOf(p.status || 'available');
    p.status = statuses[(current + 1) % statuses.length];
    localStorage.setItem('fgeneral_products', JSON.stringify(products));
    loadProducts();
    showAIMessage(`📦 "${p.name}" status changed to ${p.status.toUpperCase()}`);
}

// ===== DELETE PRODUCT =====
function deleteProduct(index) {
    if (confirm(`Delete "${products[index].name}"?`)) {
        products.splice(index, 1);
        localStorage.setItem('fgeneral_products', JSON.stringify(products));
        loadProducts();
        showAIMessage('🗑️ Product deleted.');
    }
}

// ===== SHOW SECTION =====
function showSection(section) {
    document.querySelectorAll('.dashboard-section').forEach(s => s.classList.remove('active'));
    const target = document.getElementById(section + '-section');
    if (target) target.classList.add('active');
    document.querySelectorAll('.sidebar-nav a').forEach(a => a.classList.remove('active'));
    event?.target?.classList.add('active');
}

function toggleAI() {
    const chat = document.getElementById('aiChat');
    if (chat) chat.style.display = chat.style.display === 'none' ? 'flex' : 'none';
}

function sendAIMessage() {
    const input = document.getElementById('aiInput');
    const msg = input.value.trim();
    if (!msg) return;
    input.value = '';
    showAIMessage(msg, 'user');
    processAI(msg);
}

function showAIMessage(msg, type = 'ai') {
    const container = document.getElementById('aiMessages');
    if (!container) return;

    const div = document.createElement('div');
    div.className = `ai-msg ${type === 'user' ? 'user-msg' : ''}`;
    
    if (type === 'ai') {
        div.innerHTML = `
            <span class="ai-icon"><i class="fas fa-robot"></i></span>
            <div class="msg-bubble">${msg}</div>
        `;
    } else {
        div.innerHTML = `
            <div class="msg-bubble" style="margin-left:auto; background:rgba(201,168,76,0.15);">
                ${msg}
            </div>
        `;
    }
    container.appendChild(div);
    container.scrollTop = container.scrollHeight;
}
